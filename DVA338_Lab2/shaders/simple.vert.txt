#version 410 core                                       
                                                                 
in vec3 vPos;                                                    
in vec3 vNorm;                                                   
out vec4 color;                                                  
uniform mat4 PV;    
uniform mat4 W;
                                                                 
void main(void)                                                  
{                                                                
   color = vec4(vNorm, 1.0);                                	 
   gl_Position = PV * W * vec4(vPos, 1.0f);                         
}
